\subsection{Mapeamento de canais}

\begin{itemize}[leftmargin=*]

    \item \textbf{Canais de Comunicação:} Redes Sociais com criação de conteúdo, parcerias com Clínicas Veterinárias locais, Panfletos em espaços públicos, Tráfego Pago em mecanismos de busca e redes sociais (\textit{Meta Ads} / \textit{Google Ads}) e parcerias com influenciadores digitais do nicho \textit{pet}.

    \item \textbf{Canais de Distribuição/Entrega:} Lojas oficiais de Aplicativos, \textit{Landing Page} oficial , Códigos \textit{QR} impressos em materiais físicos nas clínicas parceiras, e \textit{Links} de convite rápido via \textit{WhatsApp}.

    \item \textbf{Canais de Relacionamento:} Notificações \textit{Push} no aplicativo (lembretes de vacinas e saúde), Suporte humano interativo via \textit{WhatsApp}, Aba interna de "Ajuda/FAQ" dentro do próprio aplicativo, \textit{E-mail marketing} ou \textit{Newsletter} com foco na saúde do animal, e o Mural da Rede Solidária funcionando como comunidade colaborativa interna.
\end{itemize}
